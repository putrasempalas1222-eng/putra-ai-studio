Contoh JavaScript

const response = await fetch("https://api-mzmdqh3n6a-uc.a.run.app/chat", {
  method: "POST",
  headers: {
    "Content-Type": "application/json",
    "x-api-key": "API_KEY_KAMU"
  },
  body: JSON.stringify({
    prompt: "Halo AI"
  })
});

const data = await response.json();
console.log(data.content);

Contoh Python

import requests

url = "https://api-mzmdqh3n6a-uc.a.run.app/chat"

headers = {
    "Content-Type": "application/json",
    "x-api-key": "API_KEY_KAMU"
}

payload = {
    "prompt": "Halo AI"
}

response = requests.post(url, headers=headers, json=payload)
data = response.json()

print(data.get("content", data))
